# 1. Inicializar el proyecto
uv init proyecto

# 2. Acceder a la carpeta del proyecto
cd .\proyecto\

# 3. Instalar dependencias
uv add jupyter ipykernel

# --- OPCIÓN A: Visual Studio Code ---

# 4. Activar el entorno virtual
.\.venv\Scripts\activate.bat

# 5. Abrir Visual Studio Code
code .

# --- OPCIÓN B: Jupyter Lab ---

# 4. Registrar el entorno como kernel en Jupyter
python -m ipykernel install --user --name=mi-proyecto-jupyter --display-name "Python (uv-jupyter)"

# 5. Iniciar Jupyter Lab
uv run jupyter lab
